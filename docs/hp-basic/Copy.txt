310   SUB Wait_for_cont(OPTIONAL INTEGER Delay)
311 Wait_for_cont:  !
312   !----------------------------------------------------------------!
313   !  This subprogram will setup a key menu with only a 'Carry On'
314   !  key. If the optional parameter 'Delay' is included then this
315   !  subprogram will auto continue after the specifed number of
316   !  seconds.
317   !----------------------------------------------------------------!
318      REAL Start_delay
319      INTEGER Auto_continue
320   !----------------------------------------------------------------!
321      OFF KBD
322      OFF KEY 
323      ON KEY 5 LABEL "CarryOn",8 GOTO Carry_on
324      !
325      DISP "Press k5 ""CarryOn"" when ready to continue."
326      BEEP 2000,.1
327      BEEP 2500,.1
328      !
329      IF NPAR=1 THEN Auto_continue=ABS(Delay)
330      Start_delay=TIMEDATE
331      !
332      REPEAT
333      UNTIL Auto_continue AND (TIMEDATE-Start_delay>Auto_continue)
334      !
335 Carry_on:  !
336      DISP 
337   SUBEND

341   SUB Two_drive_copy(File_name$(*),Software$,Mass_stg_sys$)
342   ! ************************************************************** !
343      COM /Storage/ Msus_dir$,Msus$,Dest_dir$,Dest_msus$
344      INTEGER Fixed
345      Fixed=0
346   ! ************************************************************** !
347 Two_drive_copy:  !
348      OFF KEY 
349      Clr_scr
350      PRINT "**************************************************"
351      PRINT 
352      PRINT "Select the type of media you are copying the Test"
353      PRINT "Software to:"
354      PRINT 
355      PRINT "     FIXED:  Fixed Disc"
356      PRINT "     FLOPPY: 5.25 inch or 3.5 inch Floppy Disc"
357      PRINT 
358      PRINT "When copying to a Fixed Disc drive, the storage"
359      PRINT "space requirements for the Test Software is"
360      IF Software$="OP VER" THEN 
361         PRINT ">= 700 kBytes.";
362      ELSE
363         PRINT ">= 1.2 MBytes.";
364      END IF
365      PRINT " In either case, the copy which is"
366      PRINT "performed is a file by file copy to avoid"
367      PRINT "accidental erasure of files."
368      PRINT 
369      PRINT "**************************************************"
370      BEEP 
371      ON KEY 5 LABEL "FIXED" GOTO Fixed
372      ON KEY 9 LABEL "FLOPPY" GOTO Floppy
373      LOOP
374      END LOOP
375 Floppy:   !
376      OFF KEY 
377      Clr_scr
378      PRINT "**************************************************"
379      PRINT 
380      PRINT "To copy the software to another floppy, you will"
381      PRINT "need ";
382      SELECT Software$
383      CASE "PERFORMANCE"
384         SELECT Mass_stg_sys$
385         CASE "DOUBLE SIDED"
386            PRINT "2 initialized double sided 3.5 inch microfloppies."
387            PRINT "Also, the disc drive you are using for disc"
388            PRINT "initialization and copying must be a double sided"
389            PRINT "disc drive (e.g., an HP 9133H or HP 9122D)."
390            PRINT 
391            PRINT "Insert your blank microfloppies into your copy"
392            PRINT "drive and INITIALIZE the discs. Press the"
393            PRINT "keyboard's CONTINUE key when you are ready."
394            PRINT 
395            PRINT "**************************************************"
396         CASE "SINGLE SIDED"
397            PRINT "5 initialized floppies (3.5 or 5.25 inch)."
398            PRINT "INITIALIZE your blank floppies and then press the"
399            PRINT "keyboard's CONTINUE key when you are ready."
400            PRINT 
401            PRINT "**************************************************"
402         END SELECT
403      CASE "OP VER"
404         SELECT Mass_stg_sys$
405         CASE "DOUBLE SIDED"
406            PRINT "1 double sided 3.5 inch microfloppy which is"
407            PRINT "initialized for 1024 Bytes/Sector. The command for"
408            PRINT "this type of format is:"
409            PRINT "     INITIALIZE '"&Dest_msus$&"',2,3."
410            PRINT "Also, the disc drive you are using for disc"
411            PRINT "initialization and copying must be a double sided"
412            PRINT "disc drive (e.g., an HP 9133H or HP 9122D)."
413            PRINT 
414            PRINT "Insert your blank microfloppy into your copy drive"
415            PRINT "and initialize the discs. Press the keyboard's"
416            PRINT "CONTINUE key when you are ready."
417            PRINT 
418            PRINT "**************************************************"
419         CASE "SINGLE SIDED"
420            PRINT "3 initialized floppies (3.5 or 5.25 inch)."
421            PRINT "INITIALIZE your blank floppies and then press the"
422            PRINT "keyboard's CONTINUE key when you are ready."
423            PRINT 
424            PRINT "**************************************************"
425         END SELECT
426      END SELECT
427      PAUSE
428      Clr_scr
429      GOTO Copy_discs
430 Fixed:      !
431      OFF KEY 
432      Fixed=1
433 Copy_discs:      !
434      I=1
435 Error_return:      !
436      LOOP
437         ON ERROR GOTO Check_error
438         IF File_name$(I)[1,4]="DISC" THEN 
439            Get_disc(File_name$(I),"MASTER",1)
440            IF File_name$(I)="DISC 1" AND Fixed=1 THEN 
441 Create_label:  !
442               CREATE ASCII Dest_dir$&"LABEL"&Dest_msus$,1
443               ASSIGN @Lbl TO Dest_dir$&"LABEL"&Dest_msus$
444               OUTPUT @Lbl;File_name$(I),Software$,"FIXED       "
445               ASSIGN @Lbl TO *
446            END IF
447            GOTO Next_file
448         END IF
449         IF File_name$(I)="LABEL" AND Fixed=1 OR File_name$(I)="DONE" THEN GOTO Next_file
450 Copy_the_file:      !
451         DISP "Copying ";File_name$(I)
452         COPY Msus_dir$&File_name$(I)&Msus$ TO Dest_dir$&File_name$(I)&Dest_msus$
453 Next_file: !
454         I=I+1
455      EXIT IF File_name$(I)="END"
456      END LOOP
457      OFF ERROR 
458      SUBEXIT
459 Check_error:   !
460      IF ERRN=54 THEN 
461         OFF ERROR 
462         Clr_scr
463         PRINT "****************************************"
464         PRINT 
465         IF File_name$(I)[1,4]="DISC" THEN 
466            PRINT "The file LABEL already exists"
467         ELSE
468            PRINT "The file ";File_name$(I);" already exists"
469         END IF
470         PRINT "on your disc.  Do you wish to purge the"
471         PRINT "old file?"
472         PRINT 
473         PRINT "Select the appropriate softkey."
474         PRINT 
475         PRINT "****************************************"
476   !
477         IF NOT FNYes("") THEN GOTO Message
478   !
479 Purge_file:   !
480         Clr_scr
481         IF File_name$(I)[1,4]="DISC" THEN 
482            PURGE Dest_dir$&"LABEL"&Dest_msus$
483            GOTO Create_label
484         ELSE
485            PURGE Dest_dir$&File_name$(I)&Dest_msus$
486            GOTO Copy_the_file
487         END IF
488 Message:   !
489         Clr_scr
490         PRINT "**************************************************"
491         PRINT 
492         PRINT "Since you do not wish to purge the existing file,"
493         PRINT "either copy the old file onto another disc or"
494         PRINT "RENAME the old file (only if you have sufficient"
495         PRINT "disc space)."
496         PRINT 
497         PRINT "**************************************************"
498         DISP "Press the keyboard's CONTINUE key when ready."
499         PAUSE
500      ELSE
501         Clr_scr
502         PRINT "**************************************************"
503         PRINT 
504         PRINT "The program encountered an error:"
505         PRINT 
506         PRINT ERRM$
507         PRINT 
508         PRINT "**************************************************"
509         DISP "Correct the error and then press CONTINUE."
510         PAUSE
511      END IF
512      Clr_scr
513      GOTO Error_return
514 Check_error_end:   !
515 !
516   SUBEND

519 !---------------------------------------------------------------
520      COM /Storage/ Msus_dir$,Msus$,Dest_dir$,Dest_msus$
521      DIM Disc_num$[9]
522 !---------------------------------------------------------------
523 Insert_disc:   !
524      Clr_scr
525      PRINT "**************************************************"
526      PRINT 
527      IF Disc_type$="MASTER" THEN 
528         PRINT "Ensure that the Test Software's MASTER disc,"
529         PRINT Disc_number$;", is WRITE PROTECTED!"
530         PRINT 
531      END IF
532      PRINT "Insert the ";Disc_type$;" disc, ";Disc_number$&", into:"
533      PRINT 
534      PRINT "      '"&Msus_dir$&Msus$&"'."
535      PRINT 
536      IF Insert_copy=1 THEN 
537         PRINT "Insert the COPY disc into:"
538         PRINT 
539         PRINT "      '"&Dest_dir$&Dest_msus$&"'."
540         PRINT 
541      END IF
542      PRINT "**************************************************"
543      Wait_for_cont
544      Clr_scr
545      IF Disc_type$="COPY" THEN SUBEXIT
546      ASSIGN @Lbl TO Msus_dir$&"LABEL"&Msus$
547      IF NPAR=5 THEN 
548         ENTER @Lbl;Disc_num$,Software$,Mass_stg_sys$
549      ELSE
550         ENTER @Lbl;Disc_num$
551      END IF
552      ASSIGN @Lbl TO *
553      IF Disc_num$<>Disc_number$ THEN GOTO Insert_disc
554   SUBEND

556   SUB One_drive_copy(File_name$(*),Software$,Mass_stg_sys$)
557 One_drive_copy:   !
558   ! ************************************************************** !
559      COM /Storage/ Msus_dir$,Msus$,Dest_dir$,Dest_msus$
560      DIM Current_disc$[9]
561      INTEGER I,J,X
562   ! ************************************************************** !
563      OFF KEY 
564      Clr_scr
565      PRINT "**************************************************"
566      PRINT 
567      PRINT "To copy the software to another floppy, you will"
568      PRINT "need ";
569      SELECT Software$
570      CASE "PERFORMANCE"
571         SELECT Mass_stg_sys$
572         CASE "DOUBLE SIDED"
573            PRINT "2 double sided 3.5 inch microfloppies."
574            PRINT "Also, the disc drive you are using for disc"
575            PRINT "initialization and copying must be a double sided"
576            PRINT "disc drive (e.g., and HP 9133H or HP 9122D)."
577            PRINT 
578            PRINT "Insert your blank microfloppies into your copy"
579            PRINT "drive and initialize the discs. Press the"
580            PRINT "keyboard's CONTINUE key when you are ready."
581         CASE "SINGLE SIDED"
582            PRINT "5 initialized floppies (3.5 or 5.25 inch)."
583            PRINT "INITIALIZE your blank floppies and then press the"
584            PRINT "keyboard's CONTINUE key when you are ready."
585         END SELECT
586      CASE "OP VER"
587         SELECT Mass_stg_sys$
588         CASE "DOUBLE SIDED"
589            PRINT "1 double sided 3.5 inch microfloppy which is"
590            PRINT "initialized for 1024 Bytes/Sector. The command for"
591            PRINT "this type of format is:"
592            PRINT "     INITIALIZE '"&Dest_msus$&"',2,3."
593            PRINT "Also, the disc drive you are using for disc"
594            PRINT "initialization and copying must be a double sided"
595            PRINT "disc drive (e.g., an HP 9133H or HP 9122D)."
596            PRINT 
597            PRINT "Insert your blank microfloppy into your copy drive"
598            PRINT "and initialize the discs. Press the keyboard's"
599            PRINT "CONTINUE key when you are ready."
600         CASE "SINGLE SIDED"
601            PRINT "3 initialized floppies (3.5 or 5.25 inch)."
602            PRINT "INITIALIZE your blank floppies and then press the"
603            PRINT "keyboard's CONTINUE key when you are ready."
604         END SELECT
605      END SELECT
606      PRINT 
607      PRINT "NOTE:The COPY discs must be INITIALIZED and BLANK!"
608      PRINT 
609      PRINT "**************************************************"
610      PAUSE
611      Clr_scr
612      INITIALIZE ":MEMORY,0,1"
613      I=1
614      J=1
615      X=1
616 Copy_discs:      !
617      LOOP
618         ON ERROR GOTO Check_error
619         IF File_name$(I)[1,4]="DISC" THEN 
620            Get_disc(File_name$(I),"MASTER",0)
621            Current_disc$=File_name$(I)
622            GOTO Next_file
623         END IF
624         IF File_name$(I)="DONE" THEN GOTO Next_file
625 Copy_to_memory:   !
626         DISP "Copying ";File_name$(I);" to memory."
627         COPY Msus_dir$&File_name$(I)&Msus$ TO File_name$(I)&":MEMORY,0,1"
628 Next_file: !
629         I=I+1
630      EXIT IF File_name$(I)="END" OR File_name$(I)[1,4]="DISC"
631      END LOOP
632      OFF ERROR 
633      !
634 Memory_to_copy: !
635      Get_disc(Current_disc$,"COPY",0)
636      FOR J=X TO I-1
637         IF File_name$(J)[1,4]="DISC" OR File_name$(J)="DONE" THEN GOTO Next_j
638         DISP "Copying ";File_name$(J);" from memory."
639         COPY File_name$(J)&":MEMORY,0,1" TO Msus_dir$&File_name$(J)&Msus$
640 Next_j:NEXT J
641      IF File_name$(I)="END" THEN SUBEXIT
642      INITIALIZE ":MEMORY,0,1"
643      IF Mass_stg_sys$="DOUBLE SIDED" THEN 
644         Get_disc(Current_disc$,"MASTER",0)
645      END IF
646      X=I
647      GOTO Copy_discs
648         !
649 Check_error:   !
650      OFF ERROR 
651      IF ERRN=64 THEN 
652         GOTO Memory_to_copy
653      ELSE
654         Clr_scr
655         PRINT "**************************************************"
656         PRINT 
657         PRINT ERRM$
658         PRINT 
659         PRINT "**************************************************"
660         DISP "Correct the problem and then press CONTINUE."
661         PAUSE
662         GOTO Copy_discs
663      END IF
664   SUBEND
