9908 Fnpwr_mtr:DEF FNPwr_mtr(@Pwr)
9909 !
9910     Range_error=.1!..... dB 436 range-to-range error
9911     Range=0
9912     LOOP
9913        OUTPUT @Pwr;"9D+V"
9914        FOR Settle=1 TO 20
9915           WAIT (Range=73)*4
9916           ENTER @Pwr USING "X,B,X,5D,X,3D";Range,Power,Expon
9917           Pwr=Power*10^(Expon)
9918           SELECT Pwr+Range_error
9919           CASE <-19.21
9920              Rng=73
9921           CASE <-9.21
9922              Rng=74
9923           CASE <.79
9924              Rng=75
9925           CASE <10.79
9926              Rng=76
9927           CASE ELSE
9928              Rng=77
9929           END SELECT
9930           IF Range<>Rng THEN 
9931              OUTPUT @Pwr;CHR$(Rng-24)
9932              GOTO Lupe
9933           END IF
9934           IF Settle<2 THEN Lupe
9935           IF Range>74 AND Settle<7 THEN Lupe
9936           IF ABS(Power-Old_power)>1 THEN Lupe
9937           OUTPUT @Pwr;"9D+V"
9938           RETURN Pwr
9939 Lupe:     Old_power=Power
9940        NEXT Settle
9941        DISP "POWER METER NOT SETTLED !!!!!!"
9942        BEEP 1000,2
9943        PAUSE
9944     END LOOP

10328 Volt_to_pwr:DEF FNVolt_to_pwr(Voltage)
10329 !************************************************************
10330 !  This examines the log_cal data and interpolates to find
10331 !  the actual power that the 8566 is measuring based on the
10332 !  value of the passed 'Voltage'.
10333 !************************************************************
10334    COM /User_data/ U1(*),U2(*),U3(*),U4(*),Log_cal(*)
10335    Lo_guess=1
10336    Hi_guess=256
10337    LOOP
10338       SELECT Hi_guess-Lo_guess
10339       CASE 1
10340          Delta_p=Log_cal(Hi_guess,2)-Log_cal(Lo_guess,2)
10341          Delta_v=Log_cal(Hi_guess,1)-Log_cal(Lo_guess,1)
10342          Meas_del=Voltage-Log_cal(Lo_guess,1)
10343          RETURN Delta_p/Delta_v*Meas_del+Log_cal(Lo_guess,2)
10344       CASE ELSE
10345          Guess=INT((Hi_guess+Lo_guess)/2)
10346          IF Log_cal(Guess,1)>Voltage OR Log_cal(Guess,1)=0 THEN 
10347             Hi_guess=Guess
10348          ELSE
10349             Lo_guess=Guess
10350          END IF
10351       END SELECT
10352    END LOOP
10353 FNEND

10488 Set_adc:SUB Set_adc(@Dut)
10489!
10490! *****************************************
10491!      I N I T I A L I Z A T I O N
10492! *****************************************
10493    COM /Peripherals/ INTEGER Print,Kbd,Crt
10494    Is_ref_locked("Dut",Junk)
10495    IF FNDut_has("GOOD HARMON") THEN 
10496       High_freq=10
10497    ELSE
10498       High_freq=4.5
10499    END IF
10500    Low_freq=1.5
10501    INTEGER Value,Lower,Sense
10502    DIM What_it_is$[70],Cal_name$[20]
10503!
10504!
10505!=========================================
10506!        ADJUST ADC CAL CONSTANTS
10507!=========================================
10508 Adc_adjust:!
10509!
10510    OFF KEY 0
10511    Clr_scr
10512    What_it_is$="       ADC ADJUSTMENTS"
10513    OUTPUT @Dut;"IP PL0DB CW";High_freq;"GZ SHPS0DB"
10514!
10515    OUTPUT @Dut;"AM0 AM1"
10516    Target=0
10517    Cal_name$="ADC OFFSET"
10518    GOSUB Adjust_cal_con
10519!
10520    OUTPUT @Dut;"SHPS-20DB AM0 AM1"
10521    Target=-20
10522    Cal_name$="ADC GAIN HI"
10523    GOSUB Adjust_cal_con
10524!
10525    OUTPUT @Dut;"CW";Low_freq;"GZ"
10526    Target=-20
10527    Cal_name$="ADC GAIN LO"
10528    GOSUB Adjust_cal_con
10529!
10530 Exit:!------------------------------------
10531    Update_image("CHECK OFF","ADC GAIN LO","ADC GAIN HI","ADC OFFSET")
10532    OUTPUT @Dut;"IP"
10533    SUBEXIT
10534!
10535 Adjust_cal_con:!--------------------------
10536!
10537    SELECT Cal_name$!....pos or negative effect
10538    CASE "ADC OFFSET"
10539       Sense=-1
10540       Max=100
10541    CASE "ADC GAIN HI","ADC GAIN LO"
10542       Sense=1
10543       Max=100
10544    CASE ELSE
10545       DISP "huh ???"
10546       PAUSE
10547    END SELECT
10548!
10549    Steps=1
10550!
10551!
10552    Clr_scr
10553    PRINT "*************************************************"
10554    PRINT What_it_is$
10555    PRINT "*************************************************"
10556    PRINT USING "4/"
10557    PRINT "-----ADJUSTING :"&Cal_name$
10558    Incr=5
10559    Value=FNRead_cal(@Dut,FNCal_location(Cal_name$))
10560    Old_value=Value
10561!
10562 Find_straddle:!
10563    Value=FNRead_cal(@Dut,FNCal_location(Cal_name$))
10564!
10565!.....find two adjacent cal #'s
10566!     that straddle the target
10567!
10568    Down_flag=0
10569    Up_flag=0
10570    LOOP
10571       Meas=FNAdc_read(@Dut)
10572       IF ABS(Meas-Target)<.06 THEN Incr=1
10573       IF (Meas-Target)>=0 THEN 
10574          IF Down_flag THEN 
10575             Lower=Value
10576             IF Incr=1 THEN Interpolate
10577             Incr=Incr-1
10578             GOTO Find_straddle
10579          END IF
10580          Value=Value+Sense*Incr
10581          Up_flag=1
10582       END IF
10583       IF (Meas-Target)<0 THEN 
10584          IF Up_flag THEN 
10585             Lower=Value-Sense*Incr
10586             IF Incr=1 THEN Interpolate
10587             Incr=Incr-1
10588             GOTO Find_straddle
10589          END IF
10590          Value=Value-Sense*Incr
10591          Down_flag=1
10592       END IF
10593       IF ABS(Value)>Max THEN Out_of_range
10594       Write_cal(@Dut,FNCal_location(Cal_name$),Value)
10595    END LOOP
10596!
10597!
10598!  ....now double check that the target
10599!      is between Upper and Lower
10600!
10601 Interpolate:!
10602    IF ABS(Lower)>Max THEN Out_of_range
10603    Write_cal(@Dut,FNCal_location(Cal_name$),Lower)
10604    P_lower=FNAdc_read(@Dut)
10605    IF ABS(Lower+Sense)>Max THEN Out_of_range
10606    Write_cal(@Dut,FNCal_location(Cal_name$),Lower+Sense)
10607    P_upper=FNAdc_read(@Dut)
10608    IF P_upper=P_lower THEN Find_straddle
10609!
10610! ....now interpolate
10611!
10612    Best_no=Sense*(Target-P_lower)/(P_upper-P_lower)*Steps+Lower-(Steps-1)/2
10613    Value=INT(Best_no)+INT(2*(Best_no-INT(Best_no)))
10614    IF ABS(Value)>Max THEN Out_of_range
10615    Write_cal(@Dut,FNCal_location(Cal_name$),Value)
10616    PRINT 
10617    PRINT "CHANGED :";Old_value;" to ";Value
10618    WAIT 1
10619    RETURN 
10620!
10621 Out_of_range:!
10622    PRINT "CALIBRATION CONSTANT :"&Cal_name$&" OUT OF RANGE !!!!"
10623    PAUSE
10624    RETURN 
10625!
10626!
10627!
10628 SUBEND

10655!
10656!============================================

10658!

10668    OUTPUT @Dut;"IP PL0DB CW4.5GZ RP1"

10671    ON KEY 9 LABEL "ABORT" GOTO Exit

10675       IF Band<1.5 THEN Freq=Low_freq

10693                Passed=-1
10694 Im7:           IMAGE  3D.3D," GHz",3X,3D.2D,"dBm",3X,3D.2D,"dB"
10695                PRINT USING Im7;Freq,Power,Window
10696             END IF
10697          END IF
10698       NEXT Power
10699    NEXT Band
10700!
10701    BEEP 500,.5
10702    IF Passed=-1 THEN
