9011  ! *****************************************

9014  ! *****************************************
9015  !
9016  !    OUTPUT POWER ACCURACY VERIFICATION
9017  !
9018  !       part # :  08340-90068
9019  !
9020  !                 6/15/82
9021  !
9022  !
9023  ! *****************************************
9024  ! *****************************************
9025  ! Revision Log :
9026  ! 05/07/84  Added USER common block for Cal_factors(*)  ms
9027  ! 07/17/84  Added 8341 capabilities.   MS

9033  ! 03/05/86  Added Field Service FSA,FSC,FSV to SELECT Test_mode$ cases.
9034  !           Also added abort key during print out.   DG
9035  ! 03/24/86  Modified Pwracc_out to call Top_of_page after 2 plots.  DG
9036  ! 04/08/86  Modified select to determine if power level entry is within
9037  !           the allowed limits and added more operator interaction.  DG
9038  ! 05/15/86  Modified softkeys & added call to Clr_scr during graphics
9039  !           output for Series 300 compatibility.   DG
9040  ! 08/08/86  Corrected specs to reflect O&S Manual specs table.  GB/DG
9041  ! 09/08/86  Added option 003 power points to Pwracc_parms.   GB/DG
9042  ! 10/22/86  Added call to Instr_present for field.  Check value of Print
9043  !           before executing a DUMP GRAPHICS.  DG
9044  ! 11/14/86  Added Old_range to FNPwr_mtr.  MD
9045  ! 11/18/86  Changed field spec to customer spec.    DG
9046  ! 12/17/86  Changed VIEWPORT for Series 300 compatibility.  DG
9047  ! 01/07/87  Modified axis label size for Series 300 compatibility. DG
9048  ! *****************************************
9049     Rev$="A    01.07.86"!     DG
9050  ! *****************************************
9051  !
9052  !
9053  ! -----------------------------------------
9054  !      I N I T I A L I Z A T I O N
9055  ! -----------------------------------------
9056     OPTION BASE 1
9057     OFF TIMEOUT 
9058     COM /Hpib_codes/ @Dut,@Sa,@Sb,@Co,@Pwr_mtr,@Dvm,@Fgn,@Rfgn,@Het,@Mod,@Atn
9059     COM /Hpib_codes/ @Timer,@D13,@D14,@D15,@D16
9060     COM /Peripherals/ INTEGER Print,Kbd,Crt
9061     COM /User_data/ U1(*),U2(*),Cal_factors(*),U3(*),U4(*)
9062     Is_ref_locked("DUT",Junk)
9063     DIM Power(10),Freq(500)
9064     INTEGER Got_sensor_cal,Dest,High_band_zero,Low_band_zero,Ok
9065     INTEGER Store_data,Acc_fail,Flat_fail,Unlvl_print,Pwrs,Frqs,Hardcopy,Flat_f,Acc_f,Pwrptr
9066     INTEGER Auto_tracked
9067  !
9068  !
9069  !-----------------------------------------
9070  !  LOAD TEST CONFIGURATION AND SPECS
9071  !-----------------------------------------
9072     Pwracc_parms(Rev$[1,1],Opt$,Power(*),Freq(*),Pwrs,Frqs)
9073     ALLOCATE Pwr_data(0:Pwrs,0:Frqs)
9074     Max_num_pts=Frqs-1
9075  !
9076  !
9077     SELECT Test_mode$
9078     CASE "EI_","T00" TO "T99"
9079        Got_sensor_cal=1
9080        GOTO Verify
9081     CASE ELSE
9082        Got_sensor_cal=0
9083     END SELECT
9084  !
9085  ! ----------------------------------------
9086  !            TEST  MENU
9087  ! ----------------------------------------
9088 Test_menu: !
9089     Clr_scr
9090     OFF KEY 
9091     PRINT " ------------------------------------------------"
9092     PRINT "            OUTPUT POWER ACCURACY TEST"
9093     PRINT "         REV. ";Rev$;"     S/N : ";Ser_no$
9094     PRINT " ------------------------------------------------"
9095     PRINT 
9096     PRINT "o  SET UP : Test set up instructions."
9097     PRINT 
9098     PRINT "o  VERIFY : Measure Output Power Accuracy across"
9099     PRINT "            full frequency range. Stores Data."
9100     PRINT 
9101     PRINT "o  SELECT : Measure accuracy in selected frequency"
9102     PRINT "            range and power. Does not store data."
9103     PRINT 
9104     PRINT "o JANITOR : Return to Test Menu."
9105     ON KEY 0 LABEL "SET UP" GOTO Set_up
9106     ON KEY 2 LABEL "VERIFY" GOTO Verify
9107     ON KEY 4 LABEL "SELECT" GOTO Select_test
9108     ON KEY 9 LABEL "JANITOR" GOTO Exit
9109     BEEP 400,.1
9110  !
9111     DISP "------------------ select a key ------------------"
9112     LOOP
9113     END LOOP
9114  !
9115  !
9116  !-------------------------------------------
9117  !          TEST SET UP INSTRUCTIONS
9118 Set_up: !------------------------------------
9119  !
9120     Clr_scr
9121     LOCAL 7
9122     CALL Pwracc_hookup
9123     REMOTE @Pwr_mtr
9124     GOTO Test_menu
9125  !
9126  ! ----------------------------------------
9127  !           FULL & SELECT TEST
9128  ! ----------------------------------------
9129  !
9130 Verify: !-----------------------------------
9131     OFF KEY 
9132     SELECT Test_mode$
9133     CASE "T00" TO "T55"
9134        IF NOT Auto_tracked THEN 
9135           OUTPUT @Dut;"SHRP"
9136           Auto_tracked=1
9137        END IF
9138     CASE "FSP","FSC","FSV"
9139        Instr_present(@Dut,"Device Under Test")
9140        Instr_present(@Pwr_mtr,"Power Meter")
9141     END SELECT
9142     Store_data=1
9143    ! When running thermal test meter cal is already loaded
9144     IF NOT Got_sensor_cal THEN CALL Get_pwr_cal(Cal_factors(*),Got_sensor_cal)
9145     IF NOT Got_sensor_cal THEN Test_menu
9146     Pwracc_parms(Rev$[1,1],Opt$,Power(*),Freq(*),Pwrs,Frqs)
9147     GOSUB Max_min_freq
9148     Acc_fail=0
9149     Flat_fail=0
9150     GOSUB Init_data_array
9151     GOSUB Pack_freqs
9152     FOR Pwrptr=1 TO Pwrs
9153        Power_level=Power(Pwrptr)
9154        Pwr_data(Pwrptr,0)=Power_level
9155        GOSUB Measure_one_pwr
9156        Pwr_plot(Ser_no$,Opt$,Rev$,Pwr_data(*),Min_freq,Max_freq,Test_mode$,Pwrptr,Frqs,Acc_f,Flat_f,0)
9157        IF Acc_f=1 THEN Acc_fail=1
9158        IF Flat_f=1 THEN Flat_fail=1
9159     NEXT Pwrptr
9160     GOSUB Store_data
9161     GOTO Output_menu
9162  !
9163  !
9164  !
9165 Select_test: !-------------------------------
9166     SELECT Test_mode$
9167     CASE "FSP","FSV","FSC"
9168        Instr_present(@Dut,"Device Under Test")
9169        Instr_present(@Pwr_mtr,"Power Meter")
9170     END SELECT
9171     OFF KEY 
9172     ON KEY 9 LABEL "ABORT" GOTO Test_menu
9173     Acc_fail=0
9174     Store_data=0
9175     Clr_scr
9176     GOSUB Max_min_freq
9177     Get_pwr_cal(Cal_factors(*),Got_sensor_cal)
9178     IF NOT Got_sensor_cal THEN Test_menu
9179 Input_start:     !
9180     INPUT "enter START freq  (GHz) :",Startf
9181     Clr_scr
9182     IF Startf<Min_freq OR Startf>Max_freq THEN 
9183        Clr_scr
9184        PRINT "**************************************************"
9185        PRINT 
9186        PRINT "The start frequency you entered is outside of the"
9187        PRINT "frequency range of the synthesizer.  Press the Re-"
9188        PRINT "Enter softkey to re-enter the start frequency."
9189        PRINT 
9190        PRINT "**************************************************"
9191        ON KEY 5 LABEL "ReEnter" GOTO Input_start
9192        LOOP
9193        END LOOP
9194     END IF
9195 Input_stop:     !
9196     INPUT "enter STOP freq   (GHz) :",Stopf
9197     IF Stopf<Min_freq OR Stopf>Max_freq OR Stopf<=Startf THEN 
9198        Clr_scr
9199        PRINT "**************************************************"
9200        PRINT 
9201        PRINT "The stop frequency you entered is outside of the"
9202        PRINT "frequency range of the synthesizer.  Press the Re-"
9203        PRINT "Enter softkey to re-enter the start frequency."
9204        PRINT 
9205        PRINT "**************************************************"
9206        ON KEY 5 LABEL "ReEnter" GOTO Input_stop
9207        LOOP
9208        END LOOP
9209     END IF
9210 Input_points:     !
9211     INPUT "enter number of points :",Frqs
9212     IF Frqs>Max_num_pts THEN 
9213        Clr_scr
9214        PRINT "**************************************************"
9215        PRINT 
9216        PRINT "The number of points entered must be less than"
9217        PRINT VAL$(Max_num_pts)&".  Press the ReEnter softkey to re-enter"
9218        PRINT "the number of points."
9219        PRINT 
9220        PRINT "**************************************************"
9221        ON KEY 5 LABEL "ReEnter" GOTO Input_points
9222        LOOP
9223        END LOOP
9224     END IF
9225 Input_pwr:    !
9226     Max_pwr_lvl=FNMax_pwr(@Dut,Stopf,Opt$)
9227     Clr_scr
9228     PRINT "**************************************************"
9229     PRINT 
9230     PRINT "       The power level you enter must be"
9231     PRINT "       between -20 and";Max_pwr_lvl;"dBm."
9232     PRINT 
9233     PRINT "**************************************************"
9234     INPUT "enter POWER LEVEL (dBm):",Power_level
9235     IF Power_level<-20 OR Power_level>Max_pwr_lvl THEN 
9236        Clr_scr
9237        PRINT "**************************************************"
9238        PRINT 
9239        PRINT "The power level you entered is outside the allowed"
9240        PRINT "range for the selected start and stop frequencies."
9241        PRINT "The power level you enter must be between -20 and"
9242        PRINT Max_pwr_lvl;"dBm.  Press the ReEnter softkey to re-enter"
9243        PRINT "the power level."
9244        PRINT 
9245        PRINT "**************************************************"
9246        ON KEY 5 LABEL "ReEnter" GOTO Input_pwr
9247        LOOP
9248        END LOOP
9249     END IF
9250     FOR I=0 TO Frqs
9251        Freq(I+1)=Startf+I*(Stopf-Startf)/Frqs
9252     NEXT I
9253     Frqs=Frqs+1
9254     GOSUB Init_data_array
9255     GOSUB Pack_freqs
9256     Pwr_data(1,0)=Power_level
9257     Pwrptr=1
9258     Pwrs=1
9259     GOSUB Measure_one_pwr
9260     Pwr_plot(Ser_no$,Opt$,Rev$,Pwr_data(*),Startf,Stopf,Test_mode$,Pwrptr,Frqs,Acc_f,Flat_f,0)
9261     Acc_fail=Acc_f
9262     Flat_fail=Flat_f
9263     GOTO Output_menu
9264  !
9265  !
9266 Init_data_array: !...set all to -9999
9267     FOR I=0 TO Frqs
9268        FOR J=0 TO Pwrs
9269           Pwr_data(J,I)=-9999
9270        NEXT J
9271     NEXT I
9272     RETURN 
9273  !
9274 Pack_freqs: !
9275     FOR I=1 TO Frqs
9276        Pwr_data(0,I)=Freq(I)
9277     NEXT I
9278     RETURN 
9279  !
9280 Max_min_freq: !
9281     IF FNDut_has("BAND 4",Opt$) THEN 
9282        Max_freq=26.5
9283     ELSE
9284        Max_freq=20
9285     END IF
9286     Min_freq=.01
9287     RETURN 
9288  !
9289  !------------------------------------------
9290  !     MEASUREMENT SECTION
9291  !------------------------------------------
9292 Measure_one_pwr:!
9293     Clr_scr
9294     OFF KEY 
9295     ON KEY 9 LABEL "ABORT" RECOVER Test_menu
9296     OUTPUT @Dut;"IP PL0DB S3"
9297     OUTPUT @Pwr_mtr;"9D+T"
9298     OUTPUT @Dut;"IP AT0DB SHPS";Power_level;"DB S310MZS3"
9299  !
9300  !
9301 Fmt7:IMAGE 3D.3D,5X,SDD.DD
9302 Img3:IMAGE "measuring OUTPUT ACCURACY at ",SDD.D," dBm"
9303  !
9304  !
9305     Low_band_zero=0
9306     High_band_zero=0
9307     Old_range=0
9308     FOR Frqptr=1 TO Frqs
9309        DISP USING Img3;Power_level
9310        Frequency=Freq(Frqptr)
9311        OUTPUT @Dut;Frequency;"GZ"
9312        IF Frequency<2.4 THEN 
9313           IF NOT Low_band_zero THEN 
9314              OUTPUT @Dut;"RF0"
9315              Pwr_mtr_settled(@Pwr_mtr)
9316              Zero_pwr_mtr(@Pwr_mtr)
9317              Low_band_zero=1
9318              OUTPUT @Dut;"RF1"
9319              Pwr_mtr_settled(@Pwr_mtr)
9320           END IF
9321        ELSE
9322           IF NOT High_band_zero THEN 
9323              OUTPUT @Dut;"RF0"
9324              Pwr_mtr_settled(@Pwr_mtr)
9325              Zero_pwr_mtr(@Pwr_mtr)
9326              High_band_zero=1
9327              OUTPUT @Dut;"RF1"
9328              Pwr_mtr_settled(@Pwr_mtr)
9329           END IF
9330        END IF
9331        IF Power_level<=FNMax_pwr(@Dut,Frequency,Opt$) THEN 
9332           Ok=0
9333           GOSUB Check_unleveled
9334           IF Ok THEN 
9335              Meas_pwr=FNPwr_mtr(Frequency,Cal_factors(*),Old_range)
9336              PRINT USING Fmt7;Frequency,Meas_pwr
9337              Pwr_data(Pwrptr,Frqptr)=Meas_pwr
9338           END IF
9339        END IF
9340     NEXT Frqptr
9341  !
9342     RETURN 
9343  !
9344  !===========================================
9345  !-----------OUTPUT MENU SECTION-------------
9346  !===========================================
9347 Output_menu: !
9348    !
9349     SELECT Test_mode$
9350     CASE "EI_"
9351        CALL Pwracc_out(Model$,Ser_no$,Test_mode$,"PRINT",Error$)
9352        SUBEXIT
9353     CASE "T00" TO "T99"
9354        SUBEXIT
9355     END SELECT
9356    !
9357     Clr_scr
9358     OFF KEY 
9359     IF Acc_fail OR Flat_fail THEN 
9360        Result$="FAILED"
9361     ELSE
9362        Result$="PASSED"
9363     END IF
9364     PRINT "-------------------------------------------------"
9365     PRINT "      OUTPUT POWER ACCURACY TEST RESULTS :"
9366     PRINT "-------------------------------------------------"
9367     PRINT "       REV : ";Rev$;"     S/N : ";Ser_no$
9368     PRINT 
9369     PRINT "================================================="
9370     PRINT "     RESULT :   instrument ";Result$
9371     PRINT "================================================="
9372     PRINT 
9373     IF Result$="FAILED" THEN 
9374        PRINT "--- re-check FLATNESS and ALC ADJUSTMENTS !!!"
9375        PRINT 
9376     END IF
9377     PRINT "  select soft key below :"
9378     PRINT 
9379     PRINT "DISPLAY  -  Output test results to CRT."
9380     PRINT "PRINT    -  Output test results to Printer."
9381     PRINT "BEGIN    -  Return to OUTPUT POWER ACCURACY MENU."
9382     PRINT "JANITOR  -  Return to the Test Menu."
9383  !
9384     ON KEY 0 LABEL "DISPLAY" GOTO Soft_copy
9385     ON KEY 2 LABEL "PRINT" GOTO Hard_copy
9386     ON KEY 4 LABEL "BEGIN" GOTO Test_menu
9387     ON KEY 9 LABEL "JANITOR" GOTO Exit
9388  !
9389     BEEP 800,.1
9390  !
9391 Wait_4_key:GOTO Wait_4_key
9392  !
9393 Soft_copy:Dest$="DISPLAY"
9394     GOTO Print
9395  !
9396 Hard_copy:Dest$="PRINT"
9397 Print:!
9398     Pwracc_out(Model$,Ser_no$,Test_mode$,Dest$,Error$,Rev$[1,1],Opt$,Pwr_data(*),Frqs,Pwrs)
9399     GOTO Output_menu
9400  !
9401  !
9402 Exit:SUBEXIT
9403  !
9404  !
9405  !_________________________________________
9406  !
9407  !        S U B R O U T I N E S
9408  !_________________________________________
9409  !
9410  !
9411  !===========================================
9412 Store_data:  !-------------------------------
9413  !===========================================
9414     IF Acc_fail OR Flat_fail THEN 
9415        Fail_count(Pass_fail)
9416     ELSE
9417        Pass_count(Pass_fail)
9418     END IF
9419     IF Store_data THEN 
9420        Clr_scr
9421        DISP "               ***** STORING DATA *****"
9422        Pwracc_out(Model$,Ser_no$,Test_mode$,"STORE",Error$,Rev$,Opt$,Pwr_data(*),Frqs,Pwrs)
9423        DISP 
9424     END IF
9425     RETURN 
9426  !
9427  !
9428  !------------------------------------------
9429  !        CHECK FOR 8340 UNLEVELED

9431  !

9433     Peaked=0

9436     IF FNUnleveled(@Dut) THEN

9438           OUTPUT @Dut;"RP1 RP0"
9439           Peaked=1
9440           GOTO Enter_status
9441        END IF
9442        Ok=0
9443        SELECT Test_mode$
9444        CASE "CF_","EI_","FSC","FSA","FSP","FSV"
9445           Clr_scr
9446           PRINT "8340 IS UNLEVELED !!!!!!!"

9448           PRINT "correct condition and then re-run this test..."
9449           DISP "press CarryOn to return to the test menu"
9450           BEEP 1000,.1
9451           BEEP 2000,.1
9452           BEEP 700,.1
9453           BEEP 800,.1
9454           OFF KEY 
9455           Wait_for_cont
9456           GOTO Test_menu
9457        CASE ELSE
9458           IF NOT Unlvl_print THEN 
9459              Unlvl_print=1
9460              PRINTER IS Print
9461              PRINT 
9462              PRINT "**************************************************"
9463              PRINT "**************************************************"
9464              PRINT "**************************************************"
9465              PRINT 
9466              PRINT "  8340 UNLEVELED DURING POWER ACCURACY TEST !!!!"
9467              Temp$=Test_mode$[2]
9468              PRINT 
9469              PRINT "              ---- temperature : "&Temp$&" Degrees"
9470              PRINT 
9471              PRINT "**************************************************"
9472              PRINT "**************************************************"
9473              PRINT "**************************************************"
9474              PRINTER IS Crt
9475           END IF
9476        END SELECT
9477     END IF
9478     RETURN

9481 Pwracc_specs: !======================================================
9482  !
9483  !      OUTPUT ACCURACY and FLATNESS SPECIFICATIONS STORED HERE
9484  !
9485  !===================================================================
9486  SUB Pwracc_specs(Test_mode$,Opt$,REAL Power,Start(*),Stop(*),Flat_spec(*),Acc_spec(*))
9487  !
9488     DIM Slp_chg(9),Hdwr_chg(9),Meas_uncert(9),Cust_flat(9),Accy(9)
9489     INTEGER Rear,Del_atten,Spec_num
9490  !
9491  !...figure out 8340 options
9492     Del_atten=NOT FNDut_has("ATTEN",Opt$)
9493     Rear=FNDut_has("REAR OUT",Opt$)
9494  !
9495  !********************************************
9496  ! FREQUENCY BANDS for power specs ------
9497  !*******************************************
9498  !...READ frequency data
9499     RESTORE Spec_bands
9500     FOR I=1 TO 3
9501        READ Start(I),Stop(I)
9502     NEXT I
9503  !             [  start <=  BAND  < stop  ]
9504  !-----------------------------------------------------
9505  !                band A     band B      band C
9506 Spec_bands:DATA .01,2.4  ,  2.4,20  ,  20,26.500000001
9507  !*****************************************************
9508  !
9509  ! figure out which power range
9510     SELECT Power
9511     CASE >10
9512        P_range=1
9513     CASE <-10
9514        P_range=3
9515     CASE ELSE
9516        P_range=2
9517     END SELECT
9518  !
9519  !...RESTORE proper table of data
9520     IF Del_atten AND NOT Rear THEN RESTORE Spec_a1_r0
9521     IF NOT Del_atten AND NOT Rear THEN RESTORE Spec_a0_r0
9522     IF Del_atten AND Rear THEN RESTORE Spec_a1_r1
9523     IF NOT Del_atten AND Rear THEN RESTORE Spec_a0_r1
9524  !
9525  !
9526  !
9527  !
9528  !********************************************
9529  !   READ TABLE OF SPEC PARAMETERS
9530  !********************************************
9531 Read_em: !
9532     FOR Row=1 TO 5
9533        FOR I=1 TO 9
9534           SELECT Row
9535           CASE 1
9536              READ Slp_chg(I)
9537           CASE 2
9538              READ Hdwr_chg(I)
9539           CASE 3
9540              READ Meas_uncert(I)
9541           CASE 4
9542              READ Cust_flat(I)
9543           CASE 5
9544              READ Accy(I)
9545           END SELECT
9546        NEXT I
9547     NEXT Row
9548 !
9549 !***********************************************
9550 !    CALCULATE SPECS FOR THIS MODE
9551 !***********************************************
9552     FOR Spec_num=1 TO 3
9553        I=INT(Spec_num+(P_range-1)*3)
9554        SELECT Test_mode$
9555        CASE "FSC","FSA","FSP","FSV"!....CUSTOMER/FIELD SERVICE
9556           Flat_spec(Spec_num)=Cust_flat(I)
9557           Acc_spec(Spec_num)=Accy(I)
9558  !
9559        CASE "T00" TO "T99"!.....HEAT TESTING
9560           Flat_spec(Spec_num)=Cust_flat(I)-Meas_uncert(I)
9561           Acc_spec(Spec_num)=Accy(I)-Meas_uncert(I)
9562  !
9563        CASE "EI_"!.......EI
9564           Flat_spec(Spec_num)=Cust_flat(I)-Meas_uncert(I)-Hdwr_chg(I)-Slp_chg(I)/2
9565           Acc_spec(Spec_num)=Flat_spec(Spec_num)+.15
9566 !
9567        CASE "CF_"!....CAL/FINAL SPECS
9568           Flat_spec(Spec_num)=Cust_flat(I)-Meas_uncert(I)-Hdwr_chg(I)-Slp_chg(I)/2
9569           Acc_spec(Spec_num)=Flat_spec(Spec_num)+.1
9570 !
9571        END SELECT
9572     NEXT Spec_num
9573 !
9574 Spec_a0_r0: !  ************ STANDARD UNIT ***********************
9575 !          +20 to +10   |   +10  to -10   |   -10  to -20
9576 !         a    b     c  |  a     b     c  |  a     b     c
9577 !---------------------------------------------------------
9578     DATA  9  , .22 , .26 , .12 , .13 , .16 , .3  , .72 , .76!..Slp chg w/T&P
9579     DATA  9  , .13 , .16 ,  0  , .13 , .16 ,  0  , .13 , .16!..Hrwr chg w/T
9580     DATA  9  , .20 , .28 , .10 , .20 , .28 , .10 , .20 , .28!.. Meas uncerty
9581     ! Specs from Operating and Service Manual
9582     DATA  9  , 1.2 , 1.7 , .6  , 1.1 , 1.6 , .9  , 1.6 , 2.1!.. Flatness
9583     DATA  9  , 1.8 , 2.3 , .9  , 1.5 , 2.0 , 1.2 , 2.0 , 2.5!.. Accuracy
9584 !
9585 !
9586 Spec_a1_r0: ! ************** DELETE ATTENUATOR *********************
9587 !          +20 to +10   |   +10  to -10   |   -10  to -20
9588 !         a    b     c  |  a     b     c  |  a     b     c
9589 !----------------------------------------------------------
9590     DATA  9  , .22 , .26 , .12 , .13 , .16 , .3  , .72 , .76!..Slp chg w/T&P
9591     DATA  9  , .13 , .16 ,  0  , .13 , .16 ,  0  , .13 , .16!..Hrwr chg w/T
9592     DATA  9  , .20 , .28 , .10 , .20 , .28 , .10 , .20 , .28!.. Meas uncerty
9593     ! Specs from Operating and Service Manual
9594     DATA  9  , 1   , 1.4 , .6  , .9  , 1.3 , .8  , 1.5 , 1.9!.. Flatness
9595     DATA  9  , 1.6 , 2.0 , .9  , 1.3 , 1.7 , 1.7 , 2.1 , 2.5!.. Accuracy

9994 Pwr_mtr:DEF FNPwr_mtr(Freq,Cal_factors(*),OPTIONAL Old_range)
9995  !*** Added Old_range to prevent insufficient settling time if
9996  !*** range changes during frequency stepping.  ...md
9997     COM /Hpib_codes/ @Dut,@Sa,@Sb,@Co,@Pwr_mtr,@Dvm,@Fgn,@Rfgn,@Het,@Mod,@Atn
9998     COM /Hpib_codes/ @Timer,@D13,@D14,@D15,@D16
9999 Read_meter:   !
10000    OUTPUT @Pwr_mtr;"DI"
10001    ENTER @Pwr_mtr USING "X,B";Range_1
10002    IF NPAR=3 THEN 
10003       IF Old_range<>Range_1 THEN WAIT (Range_1<75)*8
10004    END IF
10005    OUTPUT @Pwr_mtr;"T"
10006    ENTER @Pwr_mtr;Pwr
10007    OUTPUT @Pwr_mtr;"I"
10008    ENTER @Pwr_mtr USING "X,B";Range_2
10009    IF ABS(Range_1-Range_2)>.01 THEN 
10010       DISP "436 meter changed ranges..."
10011       FOR I=1 TO 2
10012          WAIT (Range_2<75)*4
10013          OUTPUT @Pwr_mtr;"T"
10014          ENTER @Pwr_mtr;Junk
10015       NEXT I
10016       DISP 
10017       GOTO Read_meter
10018    END IF
10019    GOSUB Calc_cal_factor
10020    IF NPAR=3 THEN Old_range=Range_2
10021    RETURN Pwr-Correction
10022 !
10023 !----------------------------------------
10024 !     CALCULATE CAL FACTOR CORRECTION
10025 !----------------------------------------
10026 Calc_cal_factor:!
10027    Cal_no=0
10028    REPEAT
10029       Cal_no=Cal_no+1
10030    UNTIL Freq<Cal_factors(Cal_no,1) OR Cal_no=34
10031    IF Cal_no=1 AND Freq<.01 THEN 
10032       Correction=0
10033       BEEP 
10034       DISP "Frequency is less than .01 GHz in FNPwr_mtr !!!!"
10035       WAIT 1
10036       DISP 
10037       RETURN 
10038    END IF
10039    Num=Cal_factors(Cal_no,2)-Cal_factors(Cal_no-1,2)
10040    Denom=Cal_factors(Cal_no,1)-Cal_factors(Cal_no-1,1)
10041    Slope=Num/Denom
10042    Cal=Slope*(Freq-Cal_factors(Cal_no-1,1))+Cal_factors(Cal_no-1,2)
10043    Correction=10*LGT(Cal/100)
10044    RETURN

10050 SUB Pwr_plot(Ser_no$,Opt$,Rev$,Power_data(*),Fstart,Fstop,Test_mode$,INTEGER Pwrptr,Freqs,Acc_fail,Flat_fail,Slope)
10051    Slope_spec=.015!....dB/GHz   before message appears on CRT
10052    DIM Start(10),Stop(10),Flat_spec(10),Acc_spec(10),Ref_pwr(10)
10053    DIM Max_flat(10),Min_flat(10)
10054    IF Slope THEN 
10055       Power_level=0
10056    ELSE
10057       Power_level=Power_data(Pwrptr,0)
10058    END IF
10059    Pwracc_specs(Test_mode$,Opt$,Power_level,Start(*),Stop(*),Flat_spec(*),Acc_spec(*))
10060    Acc_fail=-1
10061    Flat_fail=-1
10062    Startf=Fstart
10063    Stopf=Fstop
10064    First_freq=Power_data(0,1)
10065    Last_freq=Power_data(0,Freqs)
10066    IF Power_level<-9000 THEN SUBEXIT
10067    IF First_freq>Stopf THEN SUBEXIT
10068    IF Last_freq<Startf THEN SUBEXIT
10069    IF First_freq>Startf THEN Startf=First_freq
10070    IF Last_freq<Stopf THEN Stopf=Last_freq
10071    Pwracc_specs(Test_mode$,Opt$,Power_level,Start(*),Stop(*),Flat_spec(*),Acc_spec(*))
10072    IF Slope THEN 
10073       Y2=Power_level+1
10074       Y1=Power_level-1
10075    ELSE
10076       Y2=Power_level+.5*(INT(Acc_spec(3)/.5)+2)
10077       Y1=Power_level-.5*(INT(Acc_spec(3)/.5)+2)
10078    END IF
10079    Setup_plot(Startf,Stopf,Y1,Y2)
10080    CLIP Startf,Stopf,Y1,Y2
10081    LINE TYPE 1
10082    First_flag=0
10083 !
10084 !
10085 !.........graph data, find maximum and minimum flatness points
10086    FOR I=1 TO 3
10087       Min_flat(I)=999
10088       Max_flat(I)=-999
10089    NEXT I
10090    FOR I=1 TO Freqs
10091       Pwr=Power_data(Pwrptr,I)
10092       Freq=Power_data(0,I)
10093       IF Pwr>-100 THEN 
10094          FOR X=1 TO 3
10095             IF Freq+.000000001>=Start(X) AND Freq<Stop(X) THEN 
10096                IF Pwr>Max_flat(X) THEN Max_flat(X)=Pwr
10097                IF Pwr<Min_flat(X) THEN Min_flat(X)=Pwr
10098             END IF
10099          NEXT X
10100          IF Freq+.0001>=Startf AND Freq-.0001<Stopf THEN 
10101             IF NOT First_flag THEN 
10102                MOVE Freq,Pwr
10103                First_flag=1
10104             ELSE
10105                DRAW Freq,Pwr
10106             END IF
10107          END IF
10108       END IF
10109    NEXT I
10110 !
10111    IF Min_flat(3)>900 THEN 
10112       Min_flat(3)=Min_flat(2)
10113       Max_flat(3)=Max_flat(2)
10114    END IF
10115 !
10116    IF Slope THEN !....calculate slope
10117       Sumx=0
10118       Sumxy=0
10119       Sumy=0
10120       Sumxx=0
10121       Pts=0
10122       FOR I=1 TO Freqs
10123          Pwr=Power_data(Pwrptr,I)
10124          Freq=Power_data(0,I)
10125          IF Freq+.00000000001>2.4 AND Freq<Stopf AND Pwr>-900 THEN 
10126             Sumx=Sumx+Freq
10127             Sumxy=Sumxy+Freq*Pwr
10128             Sumy=Sumy+Pwr
10129             Sumxx=Sumxx+Freq*Freq
10130             Pts=Pts+1
10131          END IF
10132       NEXT I
10133       Pwr_slope=(Sumxy-Sumx*Sumy/Pts)/(Sumxx-Sumx*Sumx/Pts)
10134    ELSE!....... draw spec lines
10135       FOR X=1 TO 3
10136          SELECT X
10137          CASE 1
10138             Ref_pwr(X)=(Max_flat(X)+Min_flat(X))/2
10139          CASE 2,3
10140             IF Flat_spec(2)-Max_flat(2)>Flat_spec(3)-Max_flat(3) THEN 
10141                Upper_space=(Flat_spec(3)+Power_level)-Max_flat(3)
10142             ELSE
10143                Upper_space=(Flat_spec(2)+Power_level)-Max_flat(2)

10362 SUB Pwr_mtr_settled(@Pwr_mtr)
10363 Pwr_mtr_settled:!
10364 !=============================================================
10365 !  This routine waits until the power meter reading is
10366 !  stable to within 4 counts for 10 consecutive readings.
10367 !=============================================================
10368 !
10369    INTEGER Give_up,Settled
10370    DISP "WAITING FOR POWER METER TO SETTLE";
10371    OUTPUT @Pwr_mtr;"9A+T"
10372    LOOP
10373       Give_up=Give_up+1
10374       DISP ".";
10375       OUTPUT @Pwr_mtr;"T"
10376       ENTER @Pwr_mtr USING "3X,4D.";Reading
10377       IF ABS(Reading-Last)<4 THEN 
10378          Settled=Settled+1
10379       ELSE
10380          Settled=0
10381          Last=Reading
10382       END IF
10383    EXIT IF Settled>10 OR Give_up>100
10384    END LOOP
10385    DISP 
10386 SUBEND
