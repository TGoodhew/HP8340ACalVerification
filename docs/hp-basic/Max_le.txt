9550           OUTPUT @Dut;"PL";Power;"DB"

9576              OUTPUT @Dut;"TSTSTS"!...SS,AUTO
9577           CASE 1
9578              WAIT .3!...CONT,AUTO
9579           CASE 2
9580              OUTPUT @Dut;"TSTS"!...SS,1 sec
9581           CASE 3
9582              OUTPUT @Dut;"TS TS TS"!...ALT SWP
9583              WAIT .1
9584           END SELECT
9585           RETURN 
9586     !

9589           SELECT Dbl_chk
9590           CASE 0!...not set yet
9591              Dbl_chk=1
9592              Power=Power+.3
9593              GOTO Set_pwr
9594           CASE 1!...increasing power to unlvl
9595              WAIT .2

9598              IF BIT(Ext,6) THEN 
9599                 Dbl_chk=2
9600                 Power=Power-.05
9601                 GOTO Set_pwr
9602              ELSE
9603                 IF Power>=20 THEN 
9604                    Power=20
9605                    GOTO Finished
9606                 END IF
9607                 Power=Power+.05
9608                 GOTO Set_pwr
9609              END IF
