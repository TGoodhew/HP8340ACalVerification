9187  SUB Cal_co_compare(@Dut,INTEGER Constants(*),All_ok)
9188 Cal_co_compare: !
9189  !---------------------------------------------------------------!
9190  !                8340A  CALIBRATION  COMPARISON
9191  !---------------------------------------------------------------!
9192  !                     Part of 08340-90088
9193  !
9194  !  Revision;  A             04/11/84              Claude F. Ashen
9195  !---------------------------------------------------------------!
9196  !  This subprogram will compare the calibration constants in the
9197  !  input array 'Constants(*)' to the calibration constants in the
9198  !  device specifed by the I/O path name '@Dut'. The output
9199  !  parameter 'All_ok' is used to return the results of the
9200  !  comparison.
9201  !---------------------------------------------------------------!
9202     COM /Peripherals/ INTEGER Print,Kbd,Crt
9203  !---------------------------------------------------------------!
9204     INTEGER I,Cal_value
9205  !---------------------------------------------------------------!
9206     All_ok=1
9207     FOR I=1 TO FNCal_location("CHECKSUM")
9208        Cal_name$=FNCal_name$(I)
9209        DISP "CHECKING CAL CONST  ";Cal_name$
9210        Cal_value=FNRead_cal(@Dut,I)
9211        !
9212        IF Cal_value<>Constants(I) THEN 
9213           All_ok=0
9214           BEEP 
9215           IF FNCal_name$(I)<>"unused" THEN 
9216              OUTPUT Print USING "K,X,DD,X,K,K";"CAL #",I,"*** ERROR ***  ",Cal_name$
9217              OUTPUT Print USING "4X,K,X,SZZZZZ,X,K,X,SZZZZZ";"CAL=",Cal_value,"IMAGE=",Constants(I)
9218              OUTPUT Print;""
9219           END IF
9220        END IF
9221        !
9222     NEXT I
9223     !
9224     DISP 
9225     !
9226  SUBEND
