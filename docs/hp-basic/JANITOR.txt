3145 SUB Load_test_file
3146 Load_test_file: !
3147  !----------------------------------------------------------------!
3148  !  This subprogram will call the test file delimiter function to
3149  !  read the name of the test file in memory. Then compair the
3150  !  name of the test file in memory to the name of the desired
3151  !  test file. If there is a match then this subprogram will end.
3152  !  If there is no match then the test file in memory is deleted
3153  !  and the appropriate test file is loaded by 'Load_sub'. Since
3154  !  there may be no test file delimiter in memory (ie. no test file
3155  !  in memory) the potential error is traped.
3156  !----------------------------------------------------------------!
3157   COM /Jan_storage/ Test$,Depend_stg$,Drivers$,Tech$,Device$
3158  !----------------------------------------------------------------!
3159   DIM Test_file_name$[12]
3160   Test_file_name$=FNTrim$(FNRead_menu_str$("FILE NAME"))
3161  !----------------------------------------------------------------!
3162   Mount_volume(Test_file_name$,"SYSTEM")
3163   ON ERROR GOSUB No_test_file
3164   WHILE Test_file_name$<>FNTest_file$
3165      OFF ERROR 
3166      DELSUB FNTest_file$ TO END
3167      CALL Load_sub(Test$,Test_file_name$,Device$)
3168   END WHILE
3169     !
3170   SUBEXIT
3171     !
3172 No_test_file:!
3173  !----------------------------------------------------------------!
3174  !  This subroutine will trap the case where there is no function
3175  !  'FNTest_file$', and load the test file. All other errors cause
3176  !  an error message to be printed. Error traping is turned off in
3177  !  this subroutine and NOT reenabled, so if the test file file
3178  !  loaded does not contain the required function 'FNTest_file$',
3179  !  the series 200 operating system error handling will cause an
3180  !  error.
3181  !----------------------------------------------------------------!
3182   OFF ERROR 
3183   IF ERRN=7 THEN 
3184      CALL Load_sub(Test$,Test_file_name$,Device$)
3185   ELSE
3186      BEEP 
3187      DISP "ERROR ";ERRN;" while trying to read test file name."
3188      PAUSE
3189   END IF
3190     !
3191   RETURN 
3192 SUBEND

4579  !----------------------------------------------------------------!

4581  !  by the input parameter '@Device'. The optional input parameter
4582  !  allows the calling context to specify the fault bits that are
4583  !  to be returned.
4584  !----------------------------------------------------------------!
4585   INTEGER Fault

4587   Instr_present(@Device,"Device Under Test")
4588   OUTPUT @Device;"OF"  !....clears FAULT byte

4593   ENTER @Device;Fault

4665  !----------------------------------------------------------------!

4667  !  for the dut. The values of the cal constants, specifed by the

4669  !  the image file. If the input parameter 'Check$'="CHECK ON"
4670  !  then both files are compared and error messages are
4671  !  displayed whenever the two files do not match. After the
4672  !  two (2) files have been compared a subprogram to check the

4676   COM /Identification/ INTEGER Station_num,Batch
4677  !----------------------------------------------------------------!
4678   COM /Hpib_codes/ @Dut,@D2,@D3,@D4,@D5,@D6,@D7,@D8,@D9,@D10

4680  !----------------------------------------------------------------!
4681   INTEGER Dut_constants(1:99),Cal_image_file(1:99)
4682   INTEGER Update_consts(1:8),File_loaded
4683   INTEGER Update_number,Compare_consts,Chk_sum_loc
4684   INTEGER Constant_value,Check_sum,I,Limits_ok
4685  !----------------------------------------------------------------!

4694  !----------------------------------------------------------------!

4699   IF NPAR>=3 THEN Update_consts(2)=FNCal_location(P2$)
4700   IF NPAR>=4 THEN Update_consts(3)=FNCal_location(P3$)
4701   IF NPAR>=5 THEN Update_consts(4)=FNCal_location(P4$)
4702   IF NPAR>=6 THEN Update_consts(5)=FNCal_location(P5$)
4703   IF NPAR>=7 THEN Update_consts(6)=FNCal_location(P6$)

4710     !

4712   Load_cal_image(Model$,Ser_num$[7,10],File_rev$,Opt$,Cal_image_file(*),File_loaded)
4713   IF File_loaded THEN

5161  !**********************************************************************!

5163  !----------------------------------------------------------------!
5164  !  This function will return a true (1) indication if the feature
5165  !  is present in the 'DUT'. If the optional parameter 'Opt$'
5166  !  is passed in it will be used to determine if the feature is
5167  !  present, otherwise the information will be read from common.

5172   INTEGER Config

5176      Config$=Opt$
5177   ELSE
5178      Config$=FNPack_option$(Option$(*))
5179   END IF
5180     !

5182    !
5183          !The check for Feature_list(1)="" is to make SURE that the
5184          !Feature_list$ gets defined the first time the program is run.
5185          !(Last_config would be 0, and an A opt 000 would have config=0)
5186     !
5187   IF Config<>Last_config OR Feature_list$(1)="" OR Last_model$<>FNModel$ THEN 
5188      GOSUB Make_list
5189      Last_config=Config
5190      Last_model$=FNModel$
5191   END IF

5210      PAUSE
5211      RETURN 0
5212   END SELECT
5213   J=1
5214   FOR I=0 TO 15
5215      READ Feature_name$,B,Logic
5216      IF (BIT(Config,B) AND Logic=1) OR (NOT (BIT(Config,B)) AND Logic=-1) THEN 
5217         Feature_list$(J)=Feature_name$
5218         J=J+1
5219      END IF
5220   NEXT I
5221   Feature_list$(J)="THAT'S ALL FOLKS"
5222   RETURN

5229   DATA  "FAST PULSE"          ,3               ,-1
5230   DATA  "GOOD PHASE NOISE"    ,4               ,-1
5231   DATA  "HI PWR"              ,5               ,1

5233   DATA  "H19"                 ,7               ,1

5236   DATA  "GOOD HARMON"         ,10              ,1
5237   DATA  "H02"                 ,11              ,1

5441  !----------------------------------------------------------------!

5443  !  by the input parameter '@Device'. The input parameter

5752 Doc_on: !
5753  !===============================================================!
5754  !  PART OF : CF_CUSTOM
5755  !---------------------------------------------------------------!
5756  !  PURPOSE : To convert the date string to a numeric value that
5757  !            can be used for comparing dates.
5758  !---------------------------------------------------------------!
5759  !  PROGRAMMER : CA                       REVISION DATE : 06/21/84
5760  !===============================================================!
5761  !  INPUT PARAMETER :
5762  !  -----------------
5763  !  Date_string$        A string variable that contains the date
5764  !                      in the form "dd MMM yy".
5765  !                      Example : "21 JUN 84"   "04 JUL 84"
5766  !
5767  !  RETURNED VALUE :
5768  !  ----------------
5769  !  A READ number that can be used to compare two dates. Output
5770  !  form is yy.ddd where 'yy' represents the year and '.ddd'
5771  !  represents the day of the year in a non sequencial form.
5772  !  EXAMPLE : "31 JAN"=.031   "01 FEB"=.033
5773  !  If the month field is not reconized then a zero will be
5774  !  returned.
5775  !===============================================================!
5776 Doc_off: !
5777  !---------------------------------------------------------------!
5778 Date_value: !(Date_strING$)
5779  !---------------------------------------------------------------!
5780   REAL Date_val
5781   DIM Month_list$[50],Date_str$[16]
5782  !---------------------------------------------------------------!
5783   DATA "JAN FEB MAR APR MAY JUN JUL AUG SEP OCT NOV DEC"
5784   RESTORE 
5785   READ Month_list$
5786     !
5787   Date_str$=FNTrim$(Date_string$)
5788   Date_val=POS(Month_list$,Date_str$[4;3])
5789   IF Date_val THEN 
5790      Date_val=(VAL(Date_str$[1])+(Date_val-1)*8)/1000
5791      Date_val=Date_val+VAL(Date_str$[8])
5792   END IF
5793     !
5794   RETURN Date_val
5795     !

7178 SUB Pwr_data(Request_pwr,Request_freq,Prog_pwr,Meas_error,INTEGER Error_code)
7179 Pwr_data: !
7180  !----------------------------------------------------------------!
7181  !             14 JUNE 1983        MS
7182  !   This subprogram examines the Power Accuracy data stored in
7183  !   the user COMMON block and returns the power error that is
7184  !   closest to the requested test power and frequency as possible.
7185  !
7186  !   Request_pwr :<INPUT>  requested test power [dBm]
7187  !   Request_freq:<INPUT>  requested frequency  [GHz]
7188  !   Prog_pwr    :<OUTPUT> closest power tested to requested power
7189  !                         [dBm]
7190  !   Meas_error  :<OUTPUT> measured 8340 error at this condition
7191  !                         [dB]
7192  !   Error_code  :<OUTPUT> defined below
7193  !    =0     data is ok
7194  !    =1     COMMON array is empty, no data returned
7195  !    =2     requested power was not measured, next closest power used
7196  !----------------------------------------------------------------!
7197   COM /Power_data/ Power_cal(*),INTEGER Num_of_freqs,Num_of_pwrs
7198   COM /Power_data/ INTEGER S_n,Data_mode,Rev_code$
7199  !----------------------------------------------------------------!
7200   Error_code=0
7201   IF Num_of_freqs=0 THEN   !.....check for empty array
7202      Error_code=1
7203      SUBEXIT
7204   END IF
7205     !
7206     !...find closest frequency
7207     !
7208   SELECT Request_freq
7209   CASE <=Power_cal(0,1)
7210      Upper_freq=1
7211      Lower_freq=1
7212   CASE >Power_cal(0,Num_of_freqs)
7213      Upper_freq=Num_of_freqs
7214      Lower_freq=Num_of_freqs
7215   CASE ELSE
7216      Upper_freq=Num_of_freqs
7217      Done=0
7218      Lower_freq=1
7219      LOOP
7220         Guess=INT((Upper_freq+Lower_freq)/2+.5)
7221         IF Power_cal(0,Guess)>Request_freq THEN 
7222            Upper_freq=Guess
7223            IF (Upper_freq-Lower_freq)<=1 THEN Done=1
7224         ELSE
7225            Lower_freq=Guess
7226            IF Upper_freq=Lower_freq THEN Done=1
7227         END IF
7228      EXIT IF Done
7229      END LOOP
7230   END SELECT
7231     !
7232     !...find closest measured power
7233     !
7234   Nearest=99999
7235   FOR Pwrptr=1 TO Num_of_pwrs
7236      IF ABS(Power_cal(Pwrptr,0)-Request_pwr)<Nearest THEN 
7237         IF Power_cal(Pwrptr,Lower_freq)>-900 AND Power_cal(Pwrptr,Upper_freq)>-900 THEN 
7238            Best_pwr=Pwrptr
7239            Nearest=ABS(Power_cal(Pwrptr,0)-Request_pwr)
7240         END IF
7241      END IF
7242   NEXT Pwrptr
7243   IF Nearest=99999 THEN 
7244      Error_code=1
7245      SUBEXIT
7246   END IF
7247   Prog_pwr=Power_cal(Best_pwr,0)
7248   IF ABS(Prog_pwr-Request_pwr)>.01 THEN Error_code=2
7249     !
7250     ! ...interpolate measured power
7251     !
7252   IF Upper_freq=Lower_freq THEN 
7253      Meas_error=Power_cal(Best_pwr,Upper_freq)-Prog_pwr
7254      SUBEXIT
7255   END IF
7256   F_ratio=(Request_freq-Power_cal(0,Lower_freq))/(Power_cal(0,Upper_freq)-Power_cal(0,Lower_freq))
7257   Meas_pwr=Power_cal(Best_pwr,Lower_freq)+F_ratio*(Power_cal(Best_pwr,Upper_freq)-Power_cal(Best_pwr,Lower_freq))
7258   Meas_error=Meas_pwr-Prog_pwr
7259     !
7260 SUBEND

7841 !================================================================!
7842 !  PART OF :   CF_LIBRARY                                        !
7843 !----------------------------------------------------------------!

7851 !                   be read.                                     !
7852 !                                                                !
7853 !    Freq           The frequency [GHz] at which the power       !
7854 !                   spec is desired.                             !
7855 !                                                                !

7995 SUB Lbl(Xpos,Ypos,Size,String$,OPTIONAL Org)
7996 Lbl: !
7997    !
7998 Look_for_delta:   !
7999   X=POS(String$,CHR$(249))
8000   IF X=0 THEN Do_it
8001   String$[X,X]="^"
8002   GOTO Look_for_delta
8003    !
8004 Do_it:   !
8005   IF NPAR>4 THEN LORG Org
8006   MOVE Xpos,Ypos
8007   CSIZE Size
8008   LABEL String$
8009   IF NPAR>4 THEN LORG 2
8010 SUBEND

8012 Pack: !
8013   WHILE POS(String$," ")
8014      String$=String$[1,POS(String$," ")-1]&String$[POS(String$," ")+1]
8015   END WHILE
8016   RETURN String$

8020 SUB Set_dvm(Desired_voltage,Tolerance)
8021 Set_dvm: !
8022  !****************************************************************
8023  !  This routine sets the 3325 to a level that produces
8024  !  'Desired_voltage' on the DVM to within 'Tolerance'
8025  !****************************************************************
8026   COM /Hpib_codes/ @Dut,@Hfsa,@Lfsa,@Cntr,@Pwrmtr,@Dvm,@Fun_gen,@Rf_gen,@Het,@Mod_an,@Atn_dvr
8027   COM /Hpib_codes/ @Timer,@D13,@D14,@D15,@D16
8028  !
8029   Prog_volts=Desired_voltage
8030   Setup_dvm(@Dvm,"IP,ACV,HOLD")
8031   REPEAT
8032      OUTPUT @Fun_gen;"AM";Prog_volts;"VR"
8033      WAIT .2
8034      Ac_rms=FNRead_dvm(@Dvm)
8035      Volt_error=Ac_rms-Desired_voltage
8036        !
8037        !..... 0.7 used to avoid 'oscillating' between two settings
8038      Prog_volts=Prog_volts-.7*Volt_error
8039   UNTIL ABS(Volt_error)<Tolerance
8040  !
8041 SUBEND

8067 Instr_present: !
8068  !************************************************************************
8069  !  This sub checks to see if an instrument being used in a test is
8070  !  present on the HP-IB.
8071  !  The pass parameter @Instrument defines the instrument's HP-IB address.

8080  !***********************************************************************

8083   STATUS @Instrument,3;Address
