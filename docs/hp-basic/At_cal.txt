10902 SUB Pwr_plot(Test_mode$,Ser_no$,Opt$,Rev$,Alc_power,Step_power(*),INTEGER Step,Raw,Acc_fail,Flat_fail)
10903    DIM Start(10),Stop(10),Flat_spec(10),Acc_spec(10),Ref_pwr(10)
10904    DIM Max_flat(10),Min_flat(10)
10905    INTEGER Ref_trace
10906    IF Step_power(0)=0 THEN SUBEXIT
10907    IF Step=0 OR Step=110 THEN Ref_trace=1
10908    Acc_fail=-1
10909    Flat_fail=-1
10910    Startf=Step_power(0)
10911    IF FNDut_has("BAND 4",Opt$) THEN 
10912       Stopf=26.5
10913    ELSE
10914       Stopf=20
10915    END IF
10916    First_freq=Step_power(0)
10917    At_cal_specs(Test_mode$,Opt$,Step,Start(*),Stop(*),Flat_spec(*),Acc_spec(*))
10918    IF Ref_trace THEN 
10919       Y2=Step_power(2)+5
10920       Y1=Step_power(2)-5
10921       Y_tic=1
10922    ELSE
10923       Y2=Power_level+.5*(INT(Acc_spec(3)/.5)+2)
10924       Y1=Power_level-.5*(INT(Acc_spec(3)/.5)+2)
10925       Y_tic=.5
10926    END IF
10927    Setup_plot(Startf,Stopf,Y1,Y2,Y_tic)
10928    CLIP Startf,Stopf,Y1,Y2
10929    LINE TYPE 1
10930    First_flag=0
10931 !
10932 !
10933 !.........graph data, find maximum and minimum flatness points
10934    FOR I=1 TO 3
10935       Min_flat(I)=999
10936       Max_flat(I)=-999
10937    NEXT I
10938    I=1
10939    LOOP
10940       Freq=Step_power(0)+(I-1)*Step_power(1)
10941    EXIT IF Freq>Stopf+.0000001
10942       Pwr=Step_power(I+1)
10943       IF Pwr>-100 THEN 
10944          FOR X=1 TO 3
10945             IF Freq>=Start(X) AND Freq<=Stop(X) THEN 
10946                IF Pwr>Max_flat(X) THEN Max_flat(X)=Pwr
10947                IF Pwr<Min_flat(X) THEN Min_flat(X)=Pwr
10948             END IF
10949          NEXT X
10950          IF Freq+.0001>=Startf AND Freq-.0001<=Stopf THEN 
10951             IF NOT First_flag THEN 
10952                MOVE Freq,Pwr
10953                First_flag=1
10954             ELSE
10955                DRAW Freq,Pwr
10956             END IF
10957          END IF
10958       END IF
10959       I=I+1
10960    END LOOP
10961 !
10962    IF NOT (Ref_trace) THEN 
10963       IF Min_flat(3)>900 THEN 
10964          Min_flat(3)=Min_flat(2)
10965          Max_flat(3)=Max_flat(2)
10966       END IF
10967 !
10968 !...calculate flatness lines
10969       FOR X=1 TO 3
10970          SELECT X
10971          CASE 1
10972             Ref_pwr(X)=(Max_flat(X)+Min_flat(X))/2
10973          CASE 2,3
10974             IF Flat_spec(2)-Max_flat(2)>Flat_spec(3)-Max_flat(3) THEN 
10975                Upper_space=Flat_spec(3)-Max_flat(3)
10976             ELSE
10977                Upper_space=Flat_spec(2)-Max_flat(2)
10978             END IF
10979             IF Min_flat(2)+Flat_spec(2)>Min_flat(3)+Flat_spec(3) THEN 
10980                Lower_space=Min_flat(3)+Flat_spec(3)
10981             ELSE
10982                Lower_space=Min_flat(2)+Flat_spec(2)
10983             END IF
10984             Ref_pwr(X)=Lower_space/2-Upper_space/2
10985          END SELECT
10986       NEXT X
10987 !....... draw spec lines
10988       FOR X=1 TO 3
10989          LINE TYPE 1
10990          MOVE Start(X),Acc_spec(X)
10991          DRAW Stop(X),Acc_spec(X)
10992          MOVE Start(X),-Acc_spec(X)
10993          DRAW Stop(X),-Acc_spec(X)
10994          LINE TYPE 4,15
10995          MOVE Start(X),Flat_spec(X)+Ref_pwr(X)
10996          DRAW Stop(X),Flat_spec(X)+Ref_pwr(X)
10997          MOVE Start(X),-Flat_spec(X)+Ref_pwr(X)
10998          DRAW Stop(X),-Flat_spec(X)+Ref_pwr(X)
10999       NEXT X
11000 !
11001 ! ....... spec the data
11002       I=1
11003       LOOP
11004          Pwr=Step_power(I+1)
11005          Freq=Step_power(0)+Step_power(1)*(I-1)
11006       EXIT IF Freq>Stopf+.000001
11007          IF Freq+.000001>=Startf AND Freq-.000001<=Stopf AND Pwr>-100 THEN 
11008             FOR X=1 TO 3
11009                IF Freq+.0001>=Start(X) AND Freq-.0001<=Stop(X) THEN 
11010                   IF ABS(Pwr-Power_level)>Acc_spec(X) THEN Acc_fail=1
11011                   IF ABS(Pwr-Ref_pwr(X))>Flat_spec(X) THEN Flat_fail=1
11012                END IF
11013             NEXT X
11014          END IF
11015          I=I+1
11016       END LOOP
11017    END IF
11018    IF Acc_fail=-1 THEN Acc_fail=0
11019    IF Flat_fail=-1 THEN Flat_fail=0
11020 !
11021 !
